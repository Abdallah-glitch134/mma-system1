// MMA System V16 — Firestore security rules
// Each authenticated player can only read/write their own fighter document.
// Deploy this rule set before production use.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /fighters/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
